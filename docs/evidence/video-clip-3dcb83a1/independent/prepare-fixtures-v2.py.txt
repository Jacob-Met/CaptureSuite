def pin(path):
    data = Path(path).read_bytes()
    return {"bytes": len(data), "sha256": hashlib.sha256(data).hexdigest(),
            "git_blob": hashlib.sha1(b"blob "+str(len(data)).encode()+b"\0"+data).hexdigest()}

def save(name, value):

def check(name, condition, details=None):

    if not condition:
        raise AssertionError(name)

def run(label, argv, *, env=None, cwd=None, expect=0, binary=False):
    started = time.perf_counter()
    proc = subprocess.Popen([str(x) for x in argv], stdout=subprocess.PIPE, stderr=subprocess.PIPE, env=env, cwd=cwd)
    try:
        out, err = proc.communicate(timeout=180)
    except subprocess.TimeoutExpired:
        proc.kill()
        out, err = proc.communicate()
        raise RuntimeError(f"{label}: timed out")
    (OUT/(label+".stdout")).write_bytes(out)
    (OUT/(label+".stderr")).write_bytes(err)
    REPORT["commands"].append({"label":label, "argv":[str(x) for x in argv], "cwd":str(cwd) if cwd else None, "pid":proc.pid, "exit":proc.returncode, "elapsed":time.perf_counter()-started, "stdout_bytes":len(out), "stderr_bytes":len(err)})
    if expect is not None:
        check(label+" exit", proc.returncode == expect, proc.returncode)
    return proc.returncode, out if binary else out.decode("utf-8", errors="replace")

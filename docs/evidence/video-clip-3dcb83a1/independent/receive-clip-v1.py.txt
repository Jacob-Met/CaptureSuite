def pin(path):
    raw=Path(path).read_bytes()
    return {"bytes":len(raw),"sha256":hashlib.sha256(raw).hexdigest(),"git_blob":hashlib.sha1(b"blob "+str(len(raw)).encode()+b"\0"+raw).hexdigest()}
def dump(path,value):
    Path(path).write_text(json.dumps(value,indent=2,ensure_ascii=False,allow_nan=False)+"\n",encoding="utf-8",newline="\n")
def tree_pins(root):
    return {p.relative_to(root).as_posix():pin(p) for p in sorted(Path(root).rglob("*")) if p.is_file()}
def check(name,condition,details=None):
    REPORT["checks"].append({"case":CURRENT,"name":name,"passed":bool(condition),"details":details})
    if not condition: raise AssertionError(name)
def run(label,argv,*,env=None,cwd=None,expected=None,binary=False):
    before=time.perf_counter()
    proc=subprocess.Popen([str(x) for x in argv],stdin=subprocess.DEVNULL,stdout=subprocess.PIPE,stderr=subprocess.PIPE,env=env,cwd=cwd)
    timed_out=False
    try: out,err=proc.communicate(timeout=180)
    except subprocess.TimeoutExpired:
        timed_out=True
        cleanup=subprocess.run(["taskkill","/PID",str(proc.pid),"/T","/F"],capture_output=True,timeout=10)
        out,err=proc.communicate(timeout=10)
        REPORT["commands"].append({"label":label+" owned timeout cleanup","exit":cleanup.returncode,"stdout":cleanup.stdout.decode(errors="replace"),"stderr":cleanup.stderr.decode(errors="replace")})
    (RESULT/(label+".stdout")).write_bytes(out)
    (RESULT/(label+".stderr")).write_bytes(err)
    REPORT["commands"].append({"label":label,"argv":[str(x) for x in argv],"cwd":str(cwd) if cwd else None,"pid":proc.pid,"exit":proc.returncode,"elapsed_seconds":time.perf_counter()-before,"stdout_bytes":len(out),"stderr_bytes":len(err),"timeout":timed_out})
    check(label+" bounded completion",not timed_out)
    if expected is not None: check(label+" exit",proc.returncode==expected,proc.returncode)
    return proc.returncode, out if binary else out.decode("utf-8",errors="replace"),err.decode("utf-8",errors="replace")
def case(name,fn):
    global CURRENT
    CURRENT=name; start=len(REPORT["checks"])
    entry={"id":name}
    try:
        fn();entry["passed"]=True
    except Exception:
        entry["passed"]=False;entry["error"]=traceback.format_exc()
    entry["assertions"]=len(REPORT["checks"])-start
    REPORT["cases"].append(entry)
def probe(label,path):
    _,text,_=run(label,[FFPROBE,"-v","error","-show_streams","-show_frames","-of","json",path],expected=0)
    return json.loads(text)
def args_for(package,out,relative,start,end):
    return [package,out,"--video-segment",relative,"--video-start-ms",str(start),"--video-end-ms",str(end)]
def cli(label,args,*,env=None):
    return run(label,[PYTHON,"-B",CANDIDATE/"tools"/"export_session.py",*args],env=env or ENV,cwd=CWD)
def refuse(label,arguments,out=None,*,env=None,late=False):
    existed=Path(out).exists() if out else False
    before=tree_pins(out) if out and Path(out).is_dir() else pin(out) if out and Path(out).is_file() else None
    status,stdout,stderr=cli(label,arguments,env=env)
    check(label+" refuses",status!=0,{"exit":status,"stdout":stdout,"stderr":stderr})
    check(label+" actionable",bool((stdout+stderr).strip()))
    if out:
        if late:
            check(label+" partial directory retained",Path(out).is_dir())
            check(label+" no completion",(not(Path(out)/"export_manifest.json").exists()))
        elif existed:
            after=tree_pins(out) if Path(out).is_dir() else pin(out)
            check(label+" existing target preserved",after==before)
        else:check(label+" no destination",not Path(out).exists())

    check(label+" exact identity",(manifest["source_id"],manifest["stream_id"],manifest["relative_path"])==(source_id,stream_id,relative))
    check(label+" exact request",manifest["requested_interval_ms"]=={"start":start,"end":end})
    source=manifest["source_video"]
    check(label+" exact source ordinal span",source["first_selected_frame"]==indices[0] and source["stop_selected_frame_exclusive"]==indices[-1]+1)
    expected_ticks=[7000+OFFSETS[index] for index in indices]
    check(label+" exact source timestamps",source["first_decoded_pts"]==7000 and Fraction(source["time_base"])==Fraction(1,1000) and source["selected_pts"]==expected_ticks)
    clip=destination/"clip.mp4"
    check(label+" output pinned",all(pin(clip)[k]==manifest["output"][k] for k in ("bytes","sha256")))
    facts=probe(label+"-probe",clip)
    streams=facts["streams"]
    check(label+" one video only",len(streams)==1 and streams[0]["codec_type"]=="video" and streams[0]["codec_name"]=="h264")
    check(label+" dimensions",(streams[0]["width"],streams[0]["height"])==(64,48))
    frames=[f for f in facts["frames"] if f.get("media_type")=="video"]
    tb=Fraction(streams[0]["time_base"])
    ticks=[int(f["best_effort_timestamp"]) for f in frames]
    check(label+" output frame timing/count",len(frames)==len(indices) and ticks[0]==0)
    check(label+" actual output provenance",manifest["output"]["frame_count"]==len(frames) and manifest["output"]["ticks"]==ticks and Fraction(manifest["output"]["time_base"])==tb)
    check(label+" relative spacing",all(abs(t*tb-Fraction(OFFSETS[i]-OFFSETS[indices[0]],1000))<=max(tb,Fraction(1,1000)) for t,i in zip(ticks,indices)))
    _,raw,_=run(label+"-decode",[FFMPEG,"-v","error","-i",clip,"-map","0:v:0","-f","rawvideo","-pix_fmt","rgb24","-fps_mode","passthrough","pipe:1"],expected=0,binary=True)
    size=64*48*3
    check(label+" decode byte count",len(raw)==size*len(indices))
    means=[]
    for position,index in enumerate(indices):
        frame=raw[position*size:(position+1)*size]
        mean=[sum(frame[channel::3])/(64*48) for channel in range(3)]
        expected=colors[index] if isinstance(colors[index],list) else [colors[index]]*3
        check(label+f" ordinal {index}",all(abs(a-b)<=8 for a,b in zip(mean,expected)),{"actual_rgb_mean":mean,"expected":expected})
        means.append(mean)
    REPORT.setdefault("clips",[]).append({"case":label,"destination":str(destination),"manifest_pin":pin(destination/"export_manifest.json"),"clip_pin":pin(clip),"indices":indices,"rgb_means":means})
